# ⚙️ 05.1 - Master python-docx Generator Template

This modular Python script serves as the production template for any assignment:

```python
import os
import subprocess
from docx import Document
from docx.shared import Pt, RGBColor, Inches
from docx.enum.text import WD_ALIGN_PARAGRAPH
from docx.enum.table import WD_TABLE_ALIGNMENT
from docx.oxml import OxmlElement, parse_xml
from docx.oxml.ns import qn, nsdecls

def create_assignment():
    doc = Document()
    
    # 1. Margins
    for sec in doc.sections:
        sec.top_margin = Inches(0.75)
        sec.bottom_margin = Inches(0.75)
        sec.left_margin = Inches(0.8)
        sec.right_margin = Inches(0.8)
        
        # 2. Outer Page Border
        sectPr = sec._sectPr
        bdr_xml = r'<w:pgBorders %s w:offsetFrom="page"><w:top w:val="single" w:sz="6" w:space="24" w:color="000000"/><w:left w:val="single" w:sz="6" w:space="24" w:color="000000"/><w:bottom w:val="single" w:sz="6" w:space="24" w:color="000000"/><w:right w:val="single" w:sz="6" w:space="24" w:color="000000"/></w:pgBorders>' % nsdecls("w")
        sectPr.append(parse_xml(bdr_xml))
        
        # 3. Footer
        footer = sec.footer
        p_f = footer.paragraphs[0]
        p_f.alignment = WD_ALIGN_PARAGRAPH.RIGHT
        r_f = p_f.add_run("Course Title | Campus Code | Page ")
        r_f.font.name = "Times New Roman"
        r_f.font.size = Pt(9)
        # Add dynamic page number XML...

    # Build Cover Page...
    # doc.add_page_break()
    # Build Content...
    
    doc.save("assignment.docx")
    subprocess.run(["libreoffice", "--headless", "--convert-to", "pdf", "assignment.docx"])

if __name__ == "__main__":
    create_assignment()
```
